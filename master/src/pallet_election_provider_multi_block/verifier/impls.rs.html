92/// A simple newtype that represents the partial backing of a winner. It only stores the total
93/// backing, and the sum of backings, as opposed to a [`sp_npos_elections::Support`] that also
94/// stores all of the backers' individual contribution.
95///
96/// This is mainly here to allow us to implement `Backings` for it.
97#[derive(Default, Encode, Decode, MaxEncodedLen, scale_info::TypeInfo)]
98pub struct PartialBackings {
99	/// The total backing of this particular winner.
100	pub total: ExtendedBalance,
101	/// The number of backers.
102	pub backers: u32,
103}
104
105impl sp_npos_elections::Backings for PartialBackings {
106	fn total(&self) -> ExtendedBalance {
107		self.total
108	}
109}

166	/// A wrapper interface for the storage items related to the queued solution.
167	///
168	/// It wraps the following:
169	///
170	/// - `QueuedSolutionX`
171	/// - `QueuedSolutionY`
172	/// - `QueuedValidVariant`
173	/// - `QueuedSolutionScore`
174	/// - `QueuedSolutionBackings`
175	///
176	/// As the name suggests, `QueuedValidVariant` points to the correct variant between
177	/// `QueuedSolutionX` and `QueuedSolutionY`. In the context of this pallet, by VALID and
178	/// INVALID variant we mean either of these two storage items, based on the value of
179	/// `QueuedValidVariant`.
180	///
181	/// ### Round Index
182	///
183	/// Much like `Snapshot` in the parent crate, these storage items are mapping whereby their
184	/// _first_ key is the round index. None of the APIs in [`QueuedSolution`] expose this, as
185	/// on-chain, we should ONLY ever be reading the current round's associated data.
186	///
187	/// Having this extra key paves the way for lazy deletion in the future.
188	///
189	/// ### Invariants
190	///
191	/// The following conditions must be met at all times for this group of storage items to be
192	/// sane.
193	///
194	/// - `QueuedSolutionScore` must always be correct. In other words, it should correctly be the
195	///   score of `QueuedValidVariant`.
196	/// - `QueuedSolutionScore` must always be [`Config::SolutionImprovementThreshold`] better than
197	///   `MinimumScore`.
198	/// - The number of existing keys in `QueuedSolutionBackings` must always match that of the
199	///   INVALID variant.
200	///
201	/// Moreover, the following conditions must be met when this pallet is in [`Status::Nothing`],
202	/// meaning that no ongoing asynchronous verification is ongoing.
203	///
204	/// - No keys should exist in the INVALID variant.
205	/// 	- This implies that no data should exist in `QueuedSolutionBackings`.
206	///
207	/// > Note that some keys *might* exist in the queued variant, but since partial solutions
208	/// > (having less than `T::Pages` pages) are in principle correct, we cannot assert anything on
209	/// > the number of keys in the VALID variant. In fact, an empty solution with score of [0, 0,
210	/// > 0] can also be correct.
211	///
212	/// No additional conditions must be met when the pallet is in [`Status::Ongoing`]. The number
213	/// of pages in
214	pub struct QueuedSolution<T: Config>(sp_std::marker::PhantomData<T>);
215	impl<T: Config> QueuedSolution<T> {
216		/// Private helper for mutating the storage group.
217		fn mutate_checked<R>(mutate: impl FnOnce() -> R) -> R {
218			let r = mutate();
219			#[cfg(debug_assertions)]
220			assert!(Self::sanity_check()
221				.inspect_err(|e| {
222					sublog!(error, "verifier", "sanity check failed: {:?}", e);
223				})
224				.is_ok());
225			r
226		}
227
228		fn round() -> u32 {
229			crate::Pallet::<T>::round()
230		}
231
232		/// Finalize a correct solution.
233		///
234		/// Should be called at the end of a verification process, once we are sure that a certain
235		/// solution is 100% correct.
236		///
237		/// It stores its score, flips the pointer to it being the current best one, and clears all
238		/// the backings and the invalid variant. (note: in principle, we can skip clearing the
239		/// backings here)
240		pub(crate) fn finalize_correct(score: ElectionScore) {
241			sublog!(
242				info,
243				"verifier",
244				"finalizing verification a correct solution, replacing old score {:?} with {:?}",
245				QueuedSolutionScore::<T>::get(Self::round()),
246				score
247			);
248
249			Self::mutate_checked(|| {
250				QueuedValidVariant::<T>::mutate(Self::round(), |v| *v = v.other());
251				QueuedSolutionScore::<T>::insert(Self::round(), score);
252
253				// Clear what was previously the valid variant. Also clears the partial backings.
254				Self::clear_invalid_and_backings_unchecked();
255			});
256		}
257
258		/// Clear all relevant information of an invalid solution.
259		///
260		/// Should be called at any step, if we encounter an issue which makes the solution
261		/// infeasible.
262		pub(crate) fn clear_invalid_and_backings() {
263			Self::mutate_checked(Self::clear_invalid_and_backings_unchecked)
264		}
265
266		/// Same as [`clear_invalid_and_backings`], but without any checks for the integrity of the
267		/// storage item group.
268		pub(crate) fn clear_invalid_and_backings_unchecked() {
269			// clear is safe as we delete at most `Pages` entries, and `Pages` is bounded.
270			match Self::invalid() {
271				ValidSolution::X => clear_round_based_map!(QueuedSolutionX::<T>, Self::round()),
272				ValidSolution::Y => clear_round_based_map!(QueuedSolutionY::<T>, Self::round()),
273			};
274			clear_round_based_map!(QueuedSolutionBackings::<T>, Self::round());
275		}
276
277		/// Write a single page of a valid solution into the `invalid` variant of the storage.
278		///
279		/// This should only be called once we are sure that this particular page is 100% correct.
280		///
281		/// This is called after *a page* has been validated, but the entire solution is not yet
282		/// known to be valid. At this stage, we write to the invalid variant. Once all pages are
283		/// verified, a call to [`finalize_correct`] will seal the correct pages and flip the
284		/// invalid/valid variants.
285		pub(crate) fn set_invalid_page(page: PageIndex, supports: SupportsOfVerifier<Pallet<T>>) {
286			use frame_support::traits::TryCollect;
287			Self::mutate_checked(|| {
288				let backings: BoundedVec<_, _> = supports
289					.iter()
290					.map(|(x, s)| (x.clone(), PartialBackings { total: s.total, backers: s.voters.len() as u32 } ))
291					.try_collect()
292					.expect("`SupportsOfVerifier` is bounded by <Pallet<T> as Verifier>::MaxWinnersPerPage, which is assured to be the same as `T::MaxWinnersPerPage` in an integrity test");
293				QueuedSolutionBackings::<T>::insert(Self::round(), page, backings);
294
295				match Self::invalid() {
296					ValidSolution::X => QueuedSolutionX::<T>::insert(Self::round(), page, supports),
297					ValidSolution::Y => QueuedSolutionY::<T>::insert(Self::round(), page, supports),
298				}
299			})
300		}
301
302		/// Write a single page to the valid variant directly.
303		///
304		/// This is not the normal flow of writing, and the solution is not checked.
305		///
306		/// This is only useful to override the valid solution with a single (likely backup)
307		/// solution.
308		pub(crate) fn force_set_single_page_valid(
309			page: PageIndex,
310			supports: SupportsOfVerifier<Pallet<T>>,
311			score: ElectionScore,
312		) {
313			Self::mutate_checked(|| {
314				// clear everything about valid solutions.
315				match Self::valid() {
316					ValidSolution::X => clear_round_based_map!(QueuedSolutionX::<T>, Self::round()),
317					ValidSolution::Y => clear_round_based_map!(QueuedSolutionY::<T>, Self::round()),
318				};
319				QueuedSolutionScore::<T>::remove(Self::round());
320
321				// write a single new page.
322				match Self::valid() {
323					ValidSolution::X => QueuedSolutionX::<T>::insert(Self::round(), page, supports),
324					ValidSolution::Y => QueuedSolutionY::<T>::insert(Self::round(), page, supports),
325				}
326				QueuedSolutionScore::<T>::insert(Self::round(), score);
327			})
328		}
329
330		pub(crate) fn force_set_multi_page_valid(
331			pages: Vec<PageIndex>,
332			supports: Vec<SupportsOfVerifier<Pallet<T>>>,
333			score: ElectionScore,
334		) {
335			debug_assert_eq!(pages.len(), supports.len());
336			// queue it in our valid queue
337			Self::mutate_checked(|| {
338				// clear everything about valid solutions.
339				match Self::valid() {
340					ValidSolution::X => clear_round_based_map!(QueuedSolutionX::<T>, Self::round()),
341					ValidSolution::Y => clear_round_based_map!(QueuedSolutionY::<T>, Self::round()),
342				};
343				QueuedSolutionScore::<T>::remove(Self::round());
344
345				// store the valid pages
346				for (support, page) in supports.into_iter().zip(pages.iter()) {
347					match Self::valid() {
348						ValidSolution::X =>
349							QueuedSolutionX::<T>::insert(Self::round(), page, support),
350						ValidSolution::Y =>
351							QueuedSolutionY::<T>::insert(Self::round(), page, support),
352					}
353				}
354				QueuedSolutionScore::<T>::insert(Self::round(), score);
355			});
356		}
357
358		/// Clear all storage items.
359		///
360		/// Should only be called once everything is done.
361		pub(crate) fn kill() {
362			Self::mutate_checked(|| {
363				clear_round_based_map!(QueuedSolutionX::<T>, Self::round());
364				clear_round_based_map!(QueuedSolutionY::<T>, Self::round());
365				QueuedValidVariant::<T>::remove(Self::round());
366				clear_round_based_map!(QueuedSolutionBackings::<T>, Self::round());
367				QueuedSolutionScore::<T>::remove(Self::round());
368			})
369		}

373		/// Return the `score` and `winner_count` of verifying solution.
374		///
375		/// Computes the final score of the solution that is currently at the end of its
376		/// verification process.
377		///
378		/// Does NOT check for completeness of all the corresponding pages of
379		/// `QueuedSolutionBackings`. This function is called during finalization logic, which can
380		/// be reached even with missing/empty pages (treated as Default::default()). Missing
381		/// pages are handled gracefully by the verification process before reaching this point.
382		/// This avoids unnecessary storage reads and redundant checks.
383		///
384		/// This solution corresponds to whatever is stored in the INVALID variant of
385		/// `QueuedSolution`. Recall that the score of this solution is not yet verified, so it
386		/// should never become `valid`.
387		pub(crate) fn compute_invalid_score() -> Result<(ElectionScore, u32), FeasibilityError> {
388			let mut total_supports: BTreeMap<T::AccountId, PartialBackings> = Default::default();
389			for (who, PartialBackings { backers, total }) in
390				QueuedSolutionBackings::<T>::iter_prefix(Self::round()).flat_map(|(_, pb)| pb)
391			{
392				let entry = total_supports.entry(who).or_default();
393				entry.total = entry.total.saturating_add(total);
394				entry.backers = entry.backers.saturating_add(backers);
395
396				if entry.backers > T::MaxBackersPerWinnerFinal::get() {
397					return Err(FeasibilityError::FailedToBoundSupport)
398				}
399			}
400
401			let winner_count = total_supports.len() as u32;
402			let score = evaluate_support(total_supports.into_values());
403
404			Ok((score, winner_count))
405		}

443		) -> impl Iterator<Item = (PageIndex, SupportsOfVerifier<Pallet<T>>)> {
444			match Self::invalid() {
445				ValidSolution::X => QueuedSolutionX::<T>::iter_prefix(Self::round()),
446				ValidSolution::Y => QueuedSolutionY::<T>::iter_prefix(Self::round()),
447			}
448		}
449
450		pub(crate) fn get_valid_page(page: PageIndex) -> Option<SupportsOfVerifier<Pallet<T>>> {
451			match Self::valid() {
452				ValidSolution::X => QueuedSolutionX::<T>::get(Self::round(), page),
453				ValidSolution::Y => QueuedSolutionY::<T>::get(Self::round(), page),
454			}
455		}

457		pub(crate) fn backing_iter() -> impl Iterator<
458			Item = (PageIndex, BoundedVec<(T::AccountId, PartialBackings), T::MaxWinnersPerPage>),
459		> {
460			QueuedSolutionBackings::<T>::iter_prefix(Self::round())
461		}

463		/// Ensure that all the storage items managed by this struct are in `kill` state, meaning
464		/// that in the expect state after an election is OVER.
465		pub(crate) fn assert_killed() {
466			use frame_support::assert_storage_noop;
467			assert_storage_noop!(Self::kill());
468		}
469
470		/// Ensure this storage item group is in correct state.
471		pub(crate) fn sanity_check() -> Result<(), sp_runtime::DispatchError> {
472			// score is correct and better than min-score.
473			ensure!(
474				Pallet::<T>::minimum_score()
475					.zip(Self::queued_score())
476					.map_or(true, |(min_score, score)| score
477						.strict_threshold_better(min_score, Perbill::zero())),
478				"queued solution has weak score (min-score)"
479			);
480
481			if let Some(queued_score) = Self::queued_score() {
482				let mut backing_map: BTreeMap<T::AccountId, PartialBackings> = BTreeMap::new();
483				Self::valid_iter()
484					.flat_map(|(_, supports)| supports)
485					.for_each(|(who, support)| {
486						let entry = backing_map.entry(who).or_default();
487						entry.total = entry.total.saturating_add(support.total);
488					});
489				let real_score = evaluate_support(backing_map.into_values());
490				ensure!(real_score == queued_score, "queued solution has wrong score");
491			} else {
492				assert!(Self::valid_iter().count() == 0, "nothing should be stored if no score");
493			}
494
495			// The number of existing keys in `QueuedSolutionBackings` must always match that of
496			// the INVALID variant.
497			ensure!(
498				QueuedSolutionBackings::<T>::iter_prefix(Self::round()).count() ==
499					Self::invalid_iter().count(),
500				"incorrect number of backings pages",
501			);
502
503			if let Status::Nothing = StatusStorage::<T>::get() {
504				ensure!(Self::invalid_iter().count() == 0, "dangling data in invalid variant");
505			}
506
507			Ok(())
508		}
509	}

513	/// The `X` variant of the current queued solution. Might be the valid one or not.
514	///
515	/// The two variants of this storage item is to avoid the need of copying. Recall that once a
516	/// `VerifyingSolution` is being processed, it needs to write its partial supports *somewhere*.
517	/// Writing theses supports on top of a *good* queued supports is wrong, since we might bail.
518	/// Writing them to a bugger and copying at the ned is slightly better, but expensive. This flag
519	/// system is best of both worlds.
520	#[pallet::storage]
521	type QueuedSolutionX<T: Config> = StorageDoubleMap<
522		_,
523		Twox64Concat,
524		u32,
525		Twox64Concat,
526		PageIndex,
527		SupportsOfVerifier<Pallet<T>>,
528	>;
529
530	/// The `Y` variant of the current queued solution. Might be the valid one or not.
531	#[pallet::storage]
532	type QueuedSolutionY<T: Config> = StorageDoubleMap<
533		_,
534		Twox64Concat,
535		u32,
536		Twox64Concat,
537		PageIndex,
538		SupportsOfVerifier<Pallet<T>>,
539	>;
540	/// Pointer to the variant of [`QueuedSolutionX`] or [`QueuedSolutionY`] that is currently
541	/// valid.
542
543	#[pallet::storage]
544	type QueuedValidVariant<T: Config> =
545		StorageMap<_, Twox64Concat, u32, ValidSolution, ValueQuery>;
546
547	/// The `(amount, count)` of backings, divided per page.
548	///
549	/// This is stored because in the last block of verification we need them to compute the score,
550	/// and check `MaxBackersPerWinnerFinal`.
551	///
552	/// This can only ever live for the invalid variant of the solution. Once it is valid, we don't
553	/// need this information anymore; the score is already computed once in
554	/// [`QueuedSolutionScore`], and the backing counts are checked.
555	#[pallet::storage]
556	type QueuedSolutionBackings<T: Config> = StorageDoubleMap<
557		_,
558		Twox64Concat,
559		u32,
560		Twox64Concat,
561		PageIndex,
562		BoundedVec<(T::AccountId, PartialBackings), T::MaxWinnersPerPage>,
563	>;

567	/// This only ever lives for the `valid` variant.

628impl<T: Config> Pallet<T> {
629	fn do_on_initialize() -> Weight {
630		if let Status::Ongoing(current_page) = Self::status_storage() {
631			let page_solution =
632				<T::SolutionDataProvider as SolutionDataProvider>::get_page(current_page);
633
634			let maybe_supports = Self::feasibility_check_page_inner(page_solution, current_page);
635
636			sublog!(
637				debug,
638				"verifier",
639				"verified page {} of a solution, outcome = {:?}",
640				current_page,
641				maybe_supports.as_ref().map(|s| s.len())
642			);
643
644			match maybe_supports {
645				Ok(supports) => {
646					Self::deposit_event(Event::<T>::Verified(current_page, supports.len() as u32));
647					QueuedSolution::<T>::set_invalid_page(current_page, supports);
648
649					if current_page > crate::Pallet::<T>::lsp() {
650						// not last page, just tick forward.
651						StatusStorage::<T>::put(Status::Ongoing(current_page.saturating_sub(1)));
652						VerifierWeightsOf::<T>::on_initialize_valid_non_terminal()
653					} else {
654						// last page, finalize everything. Get the claimed score.
655						let claimed_score = T::SolutionDataProvider::get_score();
656
657						// in both cases of the following match, we are back to the nothing state.
658						StatusStorage::<T>::put(Status::Nothing);
659
660						match Self::finalize_async_verification(claimed_score) {
661							Ok(_) => {
662								T::SolutionDataProvider::report_result(VerificationResult::Queued);
663								VerifierWeightsOf::<T>::on_initialize_valid_terminal()
664							},
665							Err(_) => {
666								T::SolutionDataProvider::report_result(
667									VerificationResult::Rejected,
668								);
669								// In case of any of the errors, kill the solution.
670								QueuedSolution::<T>::clear_invalid_and_backings();
671								VerifierWeightsOf::<T>::on_initialize_invalid_terminal()
672							},
673						}
674					}
675				},
676				Err(err) => {
677					// the page solution was invalid.
678					Self::deposit_event(Event::<T>::VerificationFailed(current_page, err));
679
680					sublog!(warn, "verifier", "Clearing any ongoing unverified solutions.");
681					// Clear any ongoing solution that has not been verified, regardless of the
682					// current state.
683					QueuedSolution::<T>::clear_invalid_and_backings_unchecked();
684
685					// we also mutate the status back to doing nothing.
686					let was_ongoing = matches!(StatusStorage::<T>::get(), Status::Ongoing(_));
687					StatusStorage::<T>::put(Status::Nothing);
688
689					if was_ongoing {
690						T::SolutionDataProvider::report_result(VerificationResult::Rejected);
691					}
692					let wasted_pages = T::Pages::get().saturating_sub(current_page);
693					VerifierWeightsOf::<T>::on_initialize_invalid_non_terminal(wasted_pages)
694				},
695			}
696		} else {
697			T::DbWeight::get().reads(1)
698		}
699	}

701	fn do_verify_synchronous_multi(
702		partial_solutions: Vec<SolutionOf<T::MinerConfig>>,
703		solution_pages: Vec<PageIndex>,
704		claimed_score: ElectionScore,
705	) -> Result<(), (PageIndex, FeasibilityError)> {
706		let first_page = solution_pages.first().cloned().unwrap_or_default();
707		let last_page = solution_pages.last().cloned().unwrap_or_default();
708		// first, ensure this score will be good enough, even if valid..
709		let _ = Self::ensure_score_quality(claimed_score).map_err(|fe| (first_page, fe))?;
710		ensure!(
711			partial_solutions.len() == solution_pages.len(),
712			(first_page, FeasibilityError::Incomplete)
713		);
714
715		// verify each page, and amalgamate into a final support.
716		let mut backings =
717			sp_std::collections::btree_map::BTreeMap::<T::AccountId, PartialBackings>::new();
718		let mut linked_supports = Vec::with_capacity(partial_solutions.len());
719
720		for (solution_page, page) in partial_solutions.into_iter().zip(solution_pages.iter()) {
721			let page_supports = Self::feasibility_check_page_inner(solution_page, *page)
722				.map_err(|fe| (*page, fe))?;
723
724			linked_supports.push(page_supports.clone());
725			let support_len = page_supports.len() as u32;
726			for (who, support) in page_supports.into_iter() {
727				let entry = backings.entry(who).or_default();
728				entry.total = entry.total.saturating_add(support.total);
729				// Note we assume snapshots are always disjoint, and therefore we can easily extend
730				// here.
731				entry.backers = entry.backers.saturating_add(support.voters.len() as u32);
732				if entry.backers > T::MaxBackersPerWinnerFinal::get() {
733					return Err((*page, FeasibilityError::FailedToBoundSupport))
734				}
735			}
736
737			Self::deposit_event(Event::<T>::Verified(*page, support_len));
738		}
739
740		// then check that the number of winners was exactly enough..
741		let desired_targets = crate::Snapshot::<T>::desired_targets()
742			.ok_or(FeasibilityError::SnapshotUnavailable)
743			.map_err(|fe| (last_page, fe))?;
744		ensure!(
745			backings.len() as u32 == desired_targets,
746			(last_page, FeasibilityError::WrongWinnerCount)
747		);
748
749		// then check the score was truth..
750		let truth_score = evaluate_support(backings.into_values());
751		ensure!(truth_score == claimed_score, (last_page, FeasibilityError::InvalidScore));
752
753		let maybe_current_score = QueuedSolution::<T>::queued_score();
754
755		// then store it.
756		sublog!(
757			debug,
758			"verifier",
759			"queued sync solution with score {:?} for pages {:?}",
760			truth_score,
761			solution_pages
762		);
763		QueuedSolution::<T>::force_set_multi_page_valid(
764			solution_pages,
765			linked_supports,
766			truth_score,
767		);
768		Self::deposit_event(Event::<T>::Queued(truth_score, maybe_current_score));
769
770		Ok(())
771	}
772
773	/// Finalize an asynchronous verification. Checks the final score for correctness, and ensures
774	/// that it matches all of the criteria.
775	///
776	/// This should only be called when all pages of an async verification are done.
777	///
778	/// Returns:
779	/// - `Ok()` if everything is okay, at which point the valid variant of the queued solution will
780	/// be updated. Returns
781	/// - `Err(Feasibility)` if any of the last verification steps fail.
782	fn finalize_async_verification(claimed_score: ElectionScore) -> Result<(), FeasibilityError> {
783		let outcome = QueuedSolution::<T>::compute_invalid_score()
784			.and_then(|(final_score, winner_count)| {
785				let desired_targets =
786					crate::Snapshot::<T>::desired_targets().defensive_unwrap_or(u32::MAX);
787				// claimed_score checked prior in seal_unverified_solution
788				match (final_score == claimed_score, winner_count == desired_targets) {
789					(true, true) => {
790						// all good, finalize this solution
791						// NOTE: must be before the call to `finalize_correct`.
792						Self::deposit_event(Event::<T>::Queued(
793							final_score,
794							QueuedSolution::<T>::queued_score(), /* the previous score, now
795							                                      * ejected. */
796						));
797						QueuedSolution::<T>::finalize_correct(final_score);
798						Ok(())
799					},
800					(false, true) => Err(FeasibilityError::InvalidScore),
801					(true, false) => Err(FeasibilityError::WrongWinnerCount),
802					(false, false) => Err(FeasibilityError::InvalidScore),
803				}
804			})
805			.map_err(|err| {
806				sublog!(warn, "verifier", "Finalizing solution was invalid due to {:?}.", err);
807				// and deposit an event about it.
808				Self::deposit_event(Event::<T>::VerificationFailed(0, err.clone()));
809				err
810			});
811		sublog!(debug, "verifier", "finalize verification outcome: {:?}", outcome);
812		outcome
813	}

815	/// Ensure that the given score is:
816	///
817	/// - better than the queued solution, if one exists.
818	/// - greater than the minimum untrusted score.
819	pub(crate) fn ensure_score_quality(score: ElectionScore) -> Result<(), FeasibilityError> {
820		let is_improvement = <Self as Verifier>::queued_score().map_or(true, |best_score| {
821			score.strict_threshold_better(best_score, T::SolutionImprovementThreshold::get())
822		});
823		ensure!(is_improvement, FeasibilityError::ScoreTooLow);
824
825		let is_greater_than_min_untrusted = Self::minimum_score()
826			.map_or(true, |min_score| score.strict_threshold_better(min_score, Perbill::zero()));
827		ensure!(is_greater_than_min_untrusted, FeasibilityError::ScoreTooLow);
828
829		Ok(())
830	}

832	/// Do the full feasibility check:
833	///
834	/// - check all edges.
835	/// - checks `MaxBackersPerWinner` to be respected IN THIS PAGE.
836	/// - checks the number of winners to be less than or equal to `DesiredTargets` IN THIS PAGE
837	///   ONLY.
838	pub(super) fn feasibility_check_page_inner(
839		partial_solution: SolutionOf<T::MinerConfig>,
840		page: PageIndex,
841	) -> Result<SupportsOfVerifier<Self>, FeasibilityError> {
842		// Read the corresponding snapshots.
843		let snapshot_targets =
844			crate::Snapshot::<T>::targets().ok_or(FeasibilityError::SnapshotUnavailable)?;
845		let snapshot_voters =
846			crate::Snapshot::<T>::voters(page).ok_or(FeasibilityError::SnapshotUnavailable)?;
847		let desired_targets =
848			crate::Snapshot::<T>::desired_targets().ok_or(FeasibilityError::SnapshotUnavailable)?;
849
850		feasibility_check_page_inner_with_snapshot::<T::MinerConfig>(
851			partial_solution,
852			&snapshot_voters,
853			&snapshot_targets,
854			desired_targets,
855		)
856		.and_then(|miner_supports| {
857			SupportsOfVerifier::<Self>::try_from_other_bounds(miner_supports)
858				.defensive_map_err(|_| FeasibilityError::FailedToBoundSupport)
859		})
860	}

868/// Same as `feasibility_check_page_inner`, but with a snapshot.
869///
870/// This is exported as a standalone function, relying on `MinerConfig` rather than `Config` so that
871/// it can be used in any offchain miner.
872pub fn feasibility_check_page_inner_with_snapshot<T: MinerConfig>(
873	partial_solution: SolutionOf<T>,
874	snapshot_voters: &BoundedVec<VoterOf<T>, T::VoterSnapshotPerBlock>,
875	snapshot_targets: &BoundedVec<T::AccountId, T::TargetSnapshotPerBlock>,
876	desired_targets: u32,
877) -> Result<PageSupportsOfMiner<T>, FeasibilityError> {
878	// ----- Start building. First, we need some closures.
879	let cache = helpers::generate_voter_cache::<T, _>(snapshot_voters);
880	let voter_at = helpers::voter_at_fn::<T>(snapshot_voters);
881	let target_at = helpers::target_at_fn::<T>(snapshot_targets);
882	let voter_index = helpers::voter_index_fn_usize::<T>(&cache);
883
884	// Then convert solution -> assignment. This will fail if any of the indices are
885	// gibberish. It will also ensure each assignemnt (voter) is unique, and all targets within it
886	// are unique.
887	let assignments = partial_solution
888		.into_assignment(voter_at, target_at)
889		.map_err::<FeasibilityError, _>(Into::into)?;
890
891	// Ensure that assignments are all correct.
892	let _ = assignments
893		.iter()
894		.map(|ref assignment| {
895			// Check that assignment.who is actually a voter (defensive-only). NOTE: while
896			// using the index map from `voter_index` is better than a blind linear search,
897			// this *still* has room for optimization. Note that we had the index when we
898			// did `solution -> assignment` and we lost it. Ideal is to keep the index
899			// around.
900
901			// Defensive-only: must exist in the snapshot.
902			let snapshot_index =
903				voter_index(&assignment.who).ok_or(FeasibilityError::InvalidVoter)?;
904			// Defensive-only: index comes from the snapshot, must exist.
905			let (_voter, _stake, targets) =
906				snapshot_voters.get(snapshot_index).ok_or(FeasibilityError::InvalidVoter)?;
907			debug_assert!(*_voter == assignment.who);
908
909			// Check that all of the targets are valid based on the snapshot.
910			if assignment.distribution.iter().any(|(t, _)| !targets.contains(t)) {
911				return Err(FeasibilityError::InvalidVote)
912			}
913			Ok(())
914		})
915		.collect::<Result<(), FeasibilityError>>()?;
916
917	// ----- Start building support. First, we need one more closure.
918	let stake_of = helpers::stake_of_fn::<T, _>(&snapshot_voters, &cache);
919
920	// This might fail if the normalization fails. Very unlikely. See `integrity_test`.
921	let staked_assignments =
922		sp_npos_elections::assignment_ratio_to_staked_normalized(assignments, stake_of)
923			.map_err::<FeasibilityError, _>(Into::into)?;
924
925	let supports = sp_npos_elections::to_supports(&staked_assignments);
926
927	// Ensure some heuristics. These conditions must hold in the **entire** support, this is
928	// just a single page. But, they must hold in a single page as well.
929	ensure!((supports.len() as u32) <= desired_targets, FeasibilityError::WrongWinnerCount);
930
931	// almost-defensive-only: `MaxBackersPerWinner` is already checked. A sane value of
932	// `MaxWinnersPerPage` should be more than any possible value of `desired_targets()`, which
933	// is ALSO checked, so this conversion can almost never fail.
934	let bounded_supports =
935		supports.try_into().map_err(|_| FeasibilityError::FailedToBoundSupport)?;
936	Ok(bounded_supports)
937}

997impl<T: Config> AsynchronousVerifier for Pallet<T> {
998	type SolutionDataProvider = T::SolutionDataProvider;
999
1000	fn status() -> Status {
1001		Pallet::<T>::status_storage()
1002	}
1003
1004	fn start() -> Result<(), &'static str> {
1005		sublog!(debug, "verifier", "start signal received.");
1006		if let Status::Nothing = Self::status() {
1007			let claimed_score = Self::SolutionDataProvider::get_score();
1008			if Self::ensure_score_quality(claimed_score).is_err() {
1009				// don't do anything, report back that this solution was garbage.
1010				Self::deposit_event(Event::<T>::VerificationFailed(
1011					crate::Pallet::<T>::msp(),
1012					FeasibilityError::ScoreTooLow,
1013				));
1014				T::SolutionDataProvider::report_result(VerificationResult::Rejected);
1015				// Despite being an instant-reject, this was a successful `start` operation.
1016				Ok(())
1017			} else {
1018				// This solution is good enough to win, we start verifying it in the next block.
1019				StatusStorage::<T>::put(Status::Ongoing(crate::Pallet::<T>::msp()));
1020				Ok(())
1021			}
1022		} else {
1023			sublog!(warn, "verifier", "start signal received while busy. This will be ignored.");
1024			Err("verification ongoing")
1025		}
1026	}
